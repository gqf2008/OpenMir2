--- 恒真性直接复算（导入真实现）---
COMPARATORS      = ['M3GUILDCSEQ', 'M3HEROPETSEQ', 'M3PAIRTRADESEQ', 'M3QUESTSEQ', 'M3SELLOFFSEQ', 'M3SHOPSEQ']
