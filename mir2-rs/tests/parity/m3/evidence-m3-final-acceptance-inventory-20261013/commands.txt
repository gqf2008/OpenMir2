## 4) ③ 技能登记/向量清点

```bash
# → inventory/30-skills-combat-registry.txt（只读；内联脚本，仓库根执行）
python -I - <<'PY'
import json
from collections import Counter
from pathlib import Path
M3 = Path('mir2-rs') / 'tests' / 'parity' / 'm3'
reg = json.loads((M3/'skill-probe'/'vectors'/'skills_registry.json').read_text(encoding='utf-8'))
rows = reg['skills']; impl = [r for r in rows if r.get('implemented')]
print('# M3 十项系统盘点 —— 技能/战斗工件计数（只读）')
print()
print('## 技能登记表（skill-probe/vectors/skills_registry.json）')
print('total_magic_rows =', reg['total_magic_rows'])
print('rows            =', len(rows))
print('implemented     =', reg['implemented'])
print('implemented_true_count =', len(impl))
print('not_implemented =', len(rows) - len(impl))
print('dospell_class_hist =', dict(Counter(r.get('dospell_class') for r in rows)))
print('implemented_magic_ids =', [r['magic_id'] for r in impl])
print('implemented_magic_ids_distinct =', len({r['magic_id'] for r in impl}))
print()
print('## 技能真身向量（skill-probe/vectors/skill_vectors.json）')
sv = json.loads((M3/'skill-probe'/'vectors'/'skill_vectors.json').read_text(encoding='utf-8'))
print('seed =', sv.get('seed'))
print('magic_hit_interval_time =', sv.get('magic_hit_interval_time'))
print('cases =', len(sv.get('cases', [])))
print('oracle keys =', list(sv.get('oracle', {}).keys()))
print()
print('## 战斗真身向量（combat-probe/vectors/combat_vectors.json）')
cv = json.loads((M3/'combat-probe'/'vectors'/'combat_vectors.json').read_text(encoding='utf-8'))
print('top keys =', list(cv.keys()))
for k, v in cv.items():
    if isinstance(v, list):
        print(' ', k, 'list len =', len(v))
PY
```
