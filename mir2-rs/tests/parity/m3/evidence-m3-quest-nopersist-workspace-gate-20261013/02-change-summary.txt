+/// 从**去注释后的源码**里抽出字符串字面量内容（标准串 + 原始串 `r"…"`/`r#"…"#`）。
+/// 目的：把「不得把任务列写进 SQL」限定在**字符串字面量**上，让展示串与注释不误报。
+fn string_literals(code: &str) -> Vec<String> {
+    let chars: Vec<char> = code.chars().collect();
+    let mut out = Vec::new();
+    let mut idx = 0usize;
+    while idx < chars.len() {
+        let cur = chars[idx];
+        // 原始串：r"…" / r#"…"# / r##"…"##（转义不生效）。
+        if cur == 'r' {
+            let mut after_r = idx + 1;
+            let mut hashes = 0usize;
+            while after_r < chars.len() && chars[after_r] == '#' {
+                hashes += 1;
+                after_r += 1;
+            }
+            if after_r < chars.len() && chars[after_r] == '"' {
+                let mut scan = after_r + 1;
+                let mut buf = String::new();
+                while scan < chars.len() {
+                    if chars[scan] == '"' {
+                        let mut matched = 0usize;
+                        while matched < hashes
+                            && scan + 1 + matched < chars.len()
+                            && chars[scan + 1 + matched] == '#'
+                        {
+                            matched += 1;
+                        }
+                        if matched == hashes {
+                            scan += 1 + hashes;
+                            break;
+                        }
+                    }
+                    buf.push(chars[scan]);
+                    scan += 1;
+                }
+                out.push(buf);
+                idx = scan;
+                continue;
+            }
+        }
+        // 字符字面量：跳过，避免 `'"'` 里的引号被当成字符串起点；生命周期 `'a` 不在此列。
+        if cur == '\'' {
+            let is_char_lit =
+                chars.get(idx + 1) == Some(&'\\') || chars.get(idx + 2) == Some(&'\'');
+            if is_char_lit {
+                let mut scan = idx + 1;
+                while scan < chars.len() {
+                    if chars[scan] == '\\' {
+                        scan += 2;
+                        continue;
+                    }
+                    if chars[scan] == '\'' {
+                        scan += 1;
+                        break;
+                    }
+                    scan += 1;
+                }
+                idx = scan;
+                continue;
+            }
+        }
+        if cur == '"' {
