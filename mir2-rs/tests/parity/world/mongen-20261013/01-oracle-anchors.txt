        public int GetMonstersZenTime(int dwTime)
        {
            if (dwTime < 30 * 60 * 1000)
            {

                {
                    if (d10 > 6)

                    return dwTime - HUtil32.Round((dwTime / 10.0) * d10);
                }
                else
                {
                    return dwTime;
                }
            }
