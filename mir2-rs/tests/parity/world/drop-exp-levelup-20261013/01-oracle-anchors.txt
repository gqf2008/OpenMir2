        private void GetExp(int dwExp)
        {
            Abil.Exp += dwExp;                                    // :151
            AddBodyLuck(dwExp * 0.002);                           // :152  ← F-2 修订：**已建模**（见 [7](a) / [9]）

            if (Abil.Exp >= Abil.MaxExp)                          // :154  ← 升级块（`if` 而非 `while`）
            {
                Abil.Exp -= Abil.MaxExp;                          // :156
                if (Abil.Level < MessageSettings.MAXUPLEVEL)      // :157  MAXUPLEVEL = byte.MaxValue = 255
                {
                    Abil.Level++;                                 // :159
                }
                HasLevelUp(Abil.Level - 1);                       // :161
                AddBodyLuck(100);                                 // :162  ← F-2 修订：**已建模**（见 [7](a) / [9]）
                IncHealthSpell(2000, 2000);                       // :164  ← HP/MP 回满 2000（按升级后 Max 截断）
            }
        }

关键口径：只要 `Abil.Exp >= Abil.MaxExp` 就进块；`Abil.Level++` 才有 `< MAXUPLEVEL` 门，
但块内 HasLevelUp / IncHealthSpell **无条件**执行 ⇒ 255 级仍要回血/重算/出帧。
（`MessageSettings.cs:7` `public const byte MAXUPLEVEL = byte.MaxValue;`）

        protected void IncHealthSpell(int nHp, int nMp)

            else { WAbil.HP += (ushort)nHp; }
            if ((WAbil.MP + nMp) >= WAbil.MaxMP) { WAbil.MP = WAbil.MaxMP; }
            else { WAbil.MP += (ushort)nMp; }
            HealthSpellChanged();
        }

        public void AddBodyLuck(double dLuck)          // src/M2Server/Player/PlayObject.cs:4304-4323
        {
            if ((dLuck > 0) && (BodyLuck < 5 * MessageSettings.BODYLUCKUNIT))
            { BodyLuck = BodyLuck + dLuck; }
            if ((dLuck < 0) && (BodyLuck > -(5 * MessageSettings.BODYLUCKUNIT)))
            { BodyLuck = BodyLuck + dLuck; }
            int n = Convert.ToInt32(BodyLuck / MessageSettings.BODYLUCKUNIT);
            if (n > 5) { n = 5; }
            if (n < -10) { n = -10; }
            BodyLuckLevel = n;
        }
