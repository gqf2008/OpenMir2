# 独立复算（不依赖 Rust 代码）：从 src/ 解析「类继承 + 无参 ctor 的 Random 调用」，
# 组合出 `CreateMonster` switch 中每条 race 的构造链抽数，与 Rust 实现逐 race 对照。
#
# 用法（cwd = <worktree>，即仓库根）：
#   python -I mir2-rs/tests/parity/world/mongen-host-20261013/40-crosscheck-ctor-draws.py.txt

# `cert.Dir`(`WorldServer.MonGen.cs:740`)，故构造链取数属生成路径 RNG 流。
import re
import glob
import sys

alltxt = [
    (f, open(f, encoding="utf-8-sig", errors="replace").read())
    for f in glob.glob("src/**/*.cs", recursive=True)
]

# class -> base（取第一个声明；BaseObject 是多文件 partial，ctor 解析时遍历所有文件）
reg = {}
for f, txt in alltxt:
    for m in re.finditer(r"\b(?:partial\s+)?class\s+(\w+)\s*(?::\s*([^\{]+))?", txt):
        base = (m.group(2) or "").strip().split(",")[0].strip()
        reg.setdefault(m.group(1), base)


def ctor_body(cls):
    """无参 ctor 的 `{...}` 体（跨全部 partial 文件查找）。"""
    for f, txt in alltxt:
        for m in re.finditer(
            r"(public|protected|private|internal)\s+" + re.escape(cls) + r"\s*\(([^)]*)\)\s*(:\s*base\s*\([^)]*\))?\s*\{",
            txt,
        ):
            if m.group(2).strip() != "":
                continue
            i = m.end() - 1
            depth = 0
            j = i
            while j < len(txt):
                if txt[j] == "{":
                    depth += 1
                elif txt[j] == "}":
                    depth -= 1
                    if depth == 0:
                        break
                j += 1
            return txt[i : j + 1]
    return ""


def draws(body):
    return re.findall(r"Random(?:Byte)?\(\s*([^\s,)]+)", body)


memo = {}


def chain(cls):
    if cls in memo:
        return memo[cls]
    base = reg.get(cls)
    res = (chain(base) if base and base != "object" else []) + draws(ctor_body(cls))
    memo[cls] = res
    return res
