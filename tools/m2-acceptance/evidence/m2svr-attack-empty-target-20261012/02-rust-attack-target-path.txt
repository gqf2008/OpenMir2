97: /// `BaseObject.GetFrontPosition`（`BaseObject.cs:977-1030`）：按 `Dir` 取前方相邻格；越界时
98: /// oracle 的 `nX/nY` **保持 CurrX/CurrY**（每个 `case` 内的边界 `if` 不满足就不动）。
99: /// 非法 `Dir`（>7）无 `default` ⇒ 同样原地。
100: #[must_use]
101: pub fn front_position(dir: u8, x: i32, y: i32, width: i32, height: i32) -> (i32, i32) {
102:     match dir {
103:         DIR_UP if y > 0 => (x, y - 1),
104:         DIR_UPRIGHT if x < width - 1 && y > 0 => (x + 1, y - 1),
105:         DIR_RIGHT if x < width - 1 => (x + 1, y),
106:         DIR_DOWNRIGHT if x < width - 1 && y < height - 1 => (x + 1, y + 1),
107:         DIR_DOWN if y < height - 1 => (x, y + 1),
108:         DIR_DOWNLEFT if x > 0 && y < height - 1 => (x - 1, y + 1),
109:         DIR_LEFT if x > 0 => (x - 1, y),
110:         DIR_UPLEFT if x > 0 && y > 0 => (x - 1, y - 1),
111:         _ => (x, y),
112:     }
113: }
